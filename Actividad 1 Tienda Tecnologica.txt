
-- Crear BD Tienda_Tecnologica
CREATE DATABASE Tienda_Tecnologica;

-- Seleciona BD
USE Tienda_Tecnologica;



-- ### TABLA: FABRICANTE ###
-- Almacena datos de los fabricantes


CREATE TABLE fabricante (
    -- Codigo unico de fabricante
    codigo INT PRIMARY KEY,

    -- Nombre del fabricante
    nombre VARCHAR(100) NOT NULL
);

-- ### TABLA: PRODUCTO ###
-- Almacena los productos de la tienda

CREATE TABLE producto (

    -- Codigo unico producto
    codigo INT PRIMARY KEY,

    -- Nombre del producto
    nombre VARCHAR(100) NOT NULL,

    -- Precio del producto con dos decimales
    precio DECIMAL(10,2) NOT NULL,

    -- Codigo del fabricante al que pertenece el producto
    codigo_fabricante INT NOT NULL,

    -- Definir codigo_fabricante como foranea
    -- para relacionarlo con la tabla fabricante
    FOREIGN KEY (codigo_fabricante)
        REFERENCES fabricante(codigo)
);
